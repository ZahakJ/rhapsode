cmake_minimum_required(VERSION 3.24)
project(rhapsode_pro LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()

option(RHAP_TESTS "Build tests" ON)
option(RHAP_APP "Build the editor (needs SDL3)" ON)

list(APPEND CMAKE_MODULE_PATH ${CMAKE_CURRENT_SOURCE_DIR}/cmake)
include(Shaders)
find_package(PkgConfig REQUIRED)
find_package(nlohmann_json 3 REQUIRED)
find_package(pugixml REQUIRED)
find_package(Threads REQUIRED)
pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil libswscale libswresample libavfilter)

add_subdirectory(third_party)
add_subdirectory(engine)
add_subdirectory(tools)
if(RHAP_APP)
  add_subdirectory(app)
endif()
if(RHAP_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
